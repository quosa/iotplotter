Connections from ESP32 to MH-Z14A

ESP PIN		-	MH PIN
-------			------
5V		-	VCC
16 (U2RXD)	-	TXD
17 (U2TXD)	-	RXD
GND		-	GND



Connections from ESP32 to BME280

ESP PIN		-	MH PIN
-------			------
3V3		-	3V3
GND		-	GND
22		-	SCL
21		-	SDA


MH-Z14A:
 - NDIR CO2 Sensor MH-Z14A PWM NDIR Infrared Carbon Dioxide Sensor Module Serial Port 0-5000PPM
 - PINS (left to right): PWM TXD RXD VCC GND ANALOG_OUTPUT HD
 - Disconnected: PWM ANALOG_OUTPUT
 - OPTIONAL: HD(brown) to ground

BME280:
 - Digital Sensor Temperature Humidity Atmospheric Pressure Sensor Module
 - PINS SDO CSB SDA/SDI SCL/SCK GND 3V3 (square solder pad)




OPTIONAL: CCS811 HDC1080 Carbon Dioxide CO2 Temperature And Humidity Sensor VOCs Air quality Monitor Sensor Module Winder

#include <Adafruit_Sensor.h>
#include <Adafruit_BME280.h>
#include <HardwareSerial.h>
#include <HTTPClient.h>
#include <WiFi.h>


#define SEALEVELPRESSURE_HPA (1013.25)

Adafruit_BME280 bme;
HardwareSerial MySerial(1);

uint8_t cmd[9] = {0xFF,0x01,0x86,0x00,0x00,0x00,0x00,0x00,0x79};
uint8_t res[9] = {};
uint8_t idx = 0;
bool flag = false;
uint16_t co2=0;

float temperature = 0;

// const char* ssid     = "bassu_";
const char* ssid     = "bassu_";
const char* password = "mer1makkara";


void setup() {
  bool status;

  Serial.begin(115200); // towards computer
  MySerial.begin(9600, SERIAL_8N1, 16, 17); // sensor
  delay(100);
 
  status = bme.begin(0x76);  
  if (!status) {
    Serial.println("Could not find a valid BME280 sensor, check wiring!");
    while (1);
  }
  
  delay(4000);
  
  Serial.print("Connecting to WiFi..");
  WiFi.begin(ssid, password); 
  while (WiFi.status() != WL_CONNECTED) { //Check for the connection
    delay(1000);
    Serial.print(".");
  }
  Serial.println();
  Serial.print("Connected to the WiFi network ");
  Serial.println(ssid);
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
  Serial.println();
 }

void loop() {
  printMHValues();
  printBMEValues();
  sendResults();
  delay(60*1000);
}

void printMHValues() {
  // Serial.println("starting");
  MySerial.write(cmd,9);
  // Serial.println("read command sent");

  while(MySerial.available()>0)
  {
    res[idx++]=MySerial.read();
    // Serial.print(res[idx -1], HEX);
    flag=true;
  }
  idx = 0;
  if(flag)
  {
    flag=false;
    co2 = 0;
    co2 += (uint16_t)res[2] <<8;
    co2 += res[3];
  }
  // Serial.println("");
  Serial.print("CO2 = ");
  Serial.print(co2);
  Serial.println(" ppm");
}

void printBMEValues() {
  temperature = bme.readTemperature();
  Serial.print("Temperature = ");
  Serial.print(temperature);
  Serial.println(" *C");
  
  // Convert temperature to Fahrenheit
  /*Serial.print("Temperature = ");
  Serial.print(1.8 * bme.readTemperature() + 32);
  Serial.println(" *F");*/
  
  Serial.print("Pressure = ");
  Serial.print(bme.readPressure() / 100.0F);
  Serial.println(" hPa");

  Serial.print("Approx. Altitude = ");
  Serial.print(bme.readAltitude(SEALEVELPRESSURE_HPA));
  Serial.println(" m");

  Serial.print("Humidity = ");
  Serial.print(bme.readHumidity());
  Serial.println(" %");

  Serial.println();
}
void sendResults() {
  if(WiFi.status() != WL_CONNECTED) {
    Serial.println("Error in WiFi connection");
    return;
  }
  
  HTTPClient http;   
 
  http.begin("http://iotplotter.com/api/v2/feed/157026536425251150");
  http.addHeader("Content-Type", "application/json");
  http.addHeader("api-key", "8c29e24f9ef569f609790bf8ed6de47a13c9756020");
 
  String post_json = "{\"data\":{\"IndoorAirQuality\": [{\"value\":";
  post_json += String(co2);
  post_json += "}],\"IndoorTemperature\": [{\"value\":";
  post_json += String(temperature);
  post_json += "}]}}";
  Serial.println(post_json);
  int httpResponseCode = http.POST(post_json);
 
  if(httpResponseCode > 0) {
    String response = http.getString();
    Serial.println(httpResponseCode);
    Serial.println(response);
   } else {
    Serial.print("Error on sending POST: ");
    Serial.println(httpResponseCode);
   }
   http.end();
}
